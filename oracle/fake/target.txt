# Double-tap: press 2 starts 100 ms after press 1's release
# (window 300 ms: double_tap_ms in plugin/src/gesture.h).

# after release; RIPL DEL / TRIM IN fire at release.

at 2800 key 0x07 down  # IN tap (deferred) -> tsel start -> cursor 1.0 (script moves it)

at 3450 key 0x08 up
